------------------------------------------------------------
Install Nodejs and NPM
------------------------------------------------------------
$ sudo apt update
$ sudo apt install -y nodejs npm

$ which node
    /usr/bin/node
$ which npm
    /usr/bin/node
$ node -v
    v22.22.1
$ npm -v
    9.2.0
------------------------------------------------------------
Install Local Http Server to server the html packages
------------------------------------------------------------
sudo npm install -g http-server
